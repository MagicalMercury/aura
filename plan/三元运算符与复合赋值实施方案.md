struct ConditionalExpr : ASTNode {
    std::unique_ptr<ASTNode> cond;
    std::unique_ptr<ASTNode> thenBranch;
    std::unique_ptr<ASTNode> elseBranch;

```cpp
    // 复合赋值脱糖：x += expr ≡ x = x + expr
    // target 允许 Identifier / obj.f（MemberAccessExpr）/ a[i]（IndexExpr），其他报错
    const char* compoundOp = nullptr;
    if (match(TokType::PlusEq))   compoundOp = "+";
    else if (match(TokType::MinusEq))  compoundOp = "-";
    else if (match(TokType::StarEq))   compoundOp = "*";
    else if (match(TokType::SlashEq))  compoundOp = "/";
    else if (match(TokType::PercentEq)) compoundOp = "%";
    if (compoundOp) {
        auto isCompoundTarget = [](const ASTNode* n) {
            return dynamic_cast<const Identifier*>(n)
                || dynamic_cast<const MemberAccessExpr*>(n)
                || dynamic_cast<const IndexExpr*>(n);
        };
        if (!isCompoundTarget(left.get())) {
            error(peek(), "compound assignment target must be an identifier, member access, or index");
            return nullptr;
        }
        auto bin = std::make_unique<BinaryExpr>();
        bin->op = compoundOp;
        bin->left = left->clone();                    // 复用 target 副本作左操作数
        bin->right = parseAssignment();               // 右结合：x -= a + b ≡ x = x - (a + b)
        auto assign = std::make_unique<AssignExpr>();
        setNodePos(assign.get(), peek());
        assign->target = std::move(left);
        assign->value = std::move(bin);
        return assign;
    }
```

**target 三类左值的 CodeGen 可行性**（已核实源码）：
- `a[i] += x` → `(*a)[i] = ((*a)[i] + x)`：[genIndexExpr](file:///d:/you/Aura/src/CodeGen/ExprGen.cpp#L1320-L1329) 生成 `(*obj)[idx]`，C++ 下标可作左值 ✅；
- `obj.f += x` → `obj->f = (obj->f + x)`：[genMemberAccess](file:///d:/you/Aura/src/CodeGen/ExprGen.cpp#L1159-L1169) 生成 `obj->field`，可作左值 ✅；
- 写屏障：target 含 `->` → [isGcFieldAssignment](file:///d:/you/Aura/src/CodeGen/ExprGen.cpp#L1448-L1452) 自动命中，字段为堆类型时生成 `gc_write_barrier`，与手写 `obj.f = obj.f + x` 一致 ✅；

**② 新增 parseConditional（放 parseAssignment 与 parsePipe 之间）**：

```cpp
std::unique_ptr<ASTNode> Parser::parseConditional() {
    auto cond = parsePipe();
    if (!cond) return nullptr;
    if (!match(TokType::Question)) return cond;

    auto thenBranch = parseExpr();      // then 是完整表达式（含赋值），':' 处自然停止
    if (!consume(TokType::Colon, "expected ':' in conditional expression"))
        return nullptr;
    auto elseBranch = parseConditional();   // 右结合
    if (!elseBranch) return nullptr;

    auto n = std::make_unique<ConditionalExpr>();
    setNodePos(n.get(), peek());
    n->cond       = std::move(cond);
    n->thenBranch = std::move(thenBranch);
    n->elseBranch = std::move(elseBranch);
    return n;
}
```

说明：`Colon` token 已存在（类型注解用），表达式上下文中 `:` 唯一用途即三元，无歧义。优先级链变为：`parseExpr → parseAssignment → parseConditional → parsePipe → ...`（C 系：三元仅高于赋值、低于 `||`/pipe）。语句级 `x += 1` 不匹配 parseExprStmt 的 `Identifier + Assign` 特判（L383-402），自然落入普通 `parseExpr` → `parseAssignment` 复合分支 ✅。

```cpp
std::unique_ptr<SemType> SemAnalyzer::inferConditional(const ConditionalExpr& e) {
    auto condTy = inferExpr(*e.cond);
    if (!isAssignable(*boolType(), *condTy)) {
        error(*e.cond, "condition of '?:' must be bool, got '" + condTy->toString() + "'");
    }
    auto thenTy = inferExpr(*e.thenBranch);
    auto elseTy = inferExpr(*e.elseBranch);
    // 单向兼容（Java JLS 15.25）：else 可赋给 then → 返回 then；否则 then 可赋给 else → 返回 else
    // bool 两分支互转成立 → 返回 bool（通用逻辑覆盖）
    // 数值提升自动生效：isAssignable(float, int)=true（§3.6）→ `cond ? 1 : 2.0` 经
    // 第二分支返回 float ✅；`cond ? 2.0 : 1` 经第一分支返回 float ✅（两分支数值混合必走对侧提升）
    if (isAssignable(*thenTy, *elseTy)) return thenTy->clone();
    if (isAssignable(*elseTy, *thenTy)) return elseTy->clone();
    error(*e.thenBranch, "incompatible '?:' branches: '" + thenTy->toString() +
          "' vs '" + elseTy->toString() + "'");
    return ErrorSemType::make();
}
```

（`isAssignable(target, source)` 为单向检查，定义于 [SemAnalyzer.cpp:357-476](file:///d:/you/Aura/src/Sema/SemAnalyzer.cpp#L357-L476)；`boolType()` 工厂在 [SemType.h:153-156](file:///d:/you/Aura/src/Sema/SemType.h#L153-L156)。）

> **数值提升在三元中的角色**：不需要给 inferConditional 单独加提升代码——`cond ? 1 : 2.0` 的判定是 `isAssignable(int, float)`（false）→ `isAssignable(float, int)`（§3.6 后为 true）→ 返回 float。分支类型规则与 Java 完全对齐：单向可赋值 + 取较宽类型。

### 3.6 语言级数值提升（用户决策：`1 + 1.0` 不报错）

**规则定义**（对齐 Java 赋值转换 JLS 5.2 + 二元数值提升 JLS 5.6.2，`%` 除外）：
- Aura 数值基元仅两个：`int`（C++ int32_t）、`float`（C++ double）。提升规则简化为一条：
  **`int` → `float` 单向加宽**（float 接受 int；int 不接受 float）。
- 赋值/初始化/返回/三元分支/二元算术中，`float` 目标可接受 `int` 源；反向（`int` 目标接 `float` 源）报错，Aura 暂无显式 cast，收窄只能靠用户写 `float` 字面量前转换。
- **`%` 保持纯整数**（用户 2026-08-08 决策，偏离 Java 浮点 `%`）：任一操作数为 float → 编译报错；int % int 返回 int。

**② [ExprInfer.cpp](file:///d:/you/Aura/src/Sema/Checker/ExprInfer.cpp) inferBinaryExpr 算术分支（L119-124）返回值修正**——原 `return lt->clone();` 只返回左操作数类型，数值混合时错误（`1 + 2.0` 应得 float 而非 int）；并在此拦截浮点 `%`：

```cpp
    // 算术：int/float（字符串拼接已在上面 "+" 分支拦截）
    if (op == "+" || op == "-" || op == "*" || op == "/" || op == "%") {
        if (!isAssignable(*lt, *rt) && !isAssignable(*rt, *lt)) {
            error(e, "binary operator '" + op + "' type mismatch: " + lt->toString() + " vs " + rt->toString());
        }
        auto* ltp = dynamic_cast<const PrimSemType*>(lt.get());
        auto* rtp = dynamic_cast<const PrimSemType*>(rt.get());
        bool lIsNum = ltp && (ltp->kind == PrimSemType::Int || ltp->kind == PrimSemType::Float);
        bool rIsNum = rtp && (rtp->kind == PrimSemType::Int || rtp->kind == PrimSemType::Float);
        if (lIsNum && rIsNum) {
            // % 保持纯整数（用户决策）：浮点取余报错；int % int → int
            if (op == "%") {
                if (ltp->kind == PrimSemType::Float || rtp->kind == PrimSemType::Float)
                    error(e, "operator '%' requires integer operands, got '" +
                          lt->toString() + "' and '" + rt->toString() + "'");
                return intType();
            }
            // 数值提升：任一操作数为 float → 结果 float；均为 int → int
            return (ltp->kind == PrimSemType::Float || rtp->kind == PrimSemType::Float)
                 ? floatType() : intType();
        }
        return lt->clone();   // 非数值基元（如 string 参与 -/* 等）：保持现有行为
    }
```

> 泛型形参 `T + T` 场景：lt/rt 为 GenericSemType，`lIsNum=false` → 回落 `lt->clone()` = T ✅（泛型算术类型推断不受影响）。

**③ CodeGen 无需改动**（`%` 纯整数后）：
- `genBinaryExpr` 兜底 `(left op right)`：`%` 仅剩 int 操作数，C++ `%` 原生支持 ✅；`int op double` 由 C++ 自动数值提升 ✅；
- 不再需要 `std::fmod` 特判，[aura_rt.h](file:///d:/you/Aura/runtime/aura_rt.h) 不新增 `<cmath>`；
- 比较运算符（L148/L152 已返回 bool，无类型检查）：`1 < 2.0` 生成 `(1 < 2.0)` ✅；
- 字符串拼接 `+`（L514-587）：字符串分支先于算术分支，不受影响 ✅。

**回归风险评估**：数值提升只放宽（加宽），不会使既有合法代码变非法。唯一行为变化是"原来编译报错的混合运算变为合法"（新功能）。既有 test.aura 中 int+int 仍返回 int，float 字面量仍是 double，无退步风险。

| 风险 | 缓解 |
|:---|:---|
